**Jacobian 推導補充**：
為了理解分母 $4(N\cdot V)(N\cdot L)$ 的來源，我們考慮微面法線 $h$ 與巨觀入射方向 $\omega_i$、出射方向 $\omega_o$ 的關係。根據鏡面反射定律，$\omega_i$ 和 $\omega_o$ 關於 $h$ 對稱。微面法線 $h$ 的立體角元 $d\omega_h$ 與巨觀入射立體角元 $d\omega_i$ 之間的關係可以通過投影幾何推導。當 $h$ 在巨觀半球上變化時，$\omega_i$ 和 $\omega_o$ 在半球上的投影面積變化不同。對於互易性 BRDF，可以證明：
$$d\omega_i = \frac{4 (N \cdot h)^2}{(N \cdot \omega_o)(N \cdot \omega_i)} d\omega_h$$
當 $h = \frac{\omega_i + \omega_o}{2}$ 且 $\omega_i, \omega_o$ 均單位化時，$(N \cdot h)^2$ 與 $(N \cdot \omega_i)(N \cdot \omega_o)$ 的關係導致了分母中的 $4$ 因子。這確保了 BRDF 在能量守恆意義下的正確尺度。

### 實作與程式（完整版）

```python
import numpy as np

def normalize(v):
    n = np.linalg.norm(v)
    if n < 1e-12:
        raise ValueError("zero-length vector")
    return v / n

def fresnel_schlick(cos_theta, f0):
    c = np.clip(cos_theta, 0.0, 1.0)
    return f0 + (1.0 - f0) * (1.0 - c) ** 5

def ggx_d(n_dot_h, roughness):
    alpha = roughness * roughness
    a2 = alpha * alpha
    denom = (n_dot_h ** 2) * (a2 - 1.0) + 1.0
    return a2 / (np.pi * denom * denom)

def smith_g1(n_dot_x, roughness):
    alpha = roughness * roughness
    a2 = alpha * alpha
    ndx = np.clip(n_dot_x, 1e-6, 1.0)
    return 2.0 * ndx / (ndx + np.sqrt(a2 + (1.0 - a2) * ndx * ndx))

def cook_torrance(N, V, L, albedo, roughness, metallic):
    N = normalize(np.asarray(N, float))
    V = normalize(np.asarray(V, float))
    L = normalize(np.asarray(L, float))
    H = normalize(V + L)
    nv = max(np.dot(N, V), 1e-6)
    nl = max(np.dot(N, L), 1e-6)
    nh = max(np.dot(N, H), 0.0)
    vh = max(np.dot(V, H), 0.0)
    f0 = 0.04 * (1.0 - metallic) + albedo * metallic
    F = fresnel_schlick(vh, f0)
    D = ggx_d(nh, roughness)
    G = smith_g1(nv, roughness) * smith_g1(nl, roughness)
    spec = F * D * G / (4.0 * nv * nl)
    kd = (1.0 - F) * (1.0 - metallic)
    diff = kd * albedo / np.pi
    return diff + spec, diff, spec

def run_tests():
    print("--- 執行測試 ---")
    
    # 測試 1: 正向 GGX 峰值
    d_peak = ggx_d(1.0, 0.5)
    expected_d = 5.093
    assert abs(d_peak - expected_d) < 0.01, f"GGX peak failed: {d_peak}"
    print(f"Test 1 (GGX Peak): PASSED (Expected ~{expected_d}, Got {d_peak:.4f})")

    # 測試 2: Schlick 掠射極限
    f_grazing = fresnel_schlick(0.0, 0.04)
    assert abs(f_grazing - 1.0) < 1e-6, f"Fresnel grazing failed: {f_grazing}"
    print(f"Test 2 (Fresnel Grazing): PASSED (Expected 1.0, Got {f_grazing:.4f})")

    # 測試 3: 金屬切換
    total, diff, spec = cook_torrance([0,0,1], [0,0,1], [1,0,1], 0.5, 0.5, 1.0)
    assert diff < 1e-6, f"Metallic diffuse failed: {diff}"
    print(f"Test 3 (Metallic Diffuse): PASSED (Diff = {diff:.6f})")

    # 測試 4: 例題 16.2 一致性
    N = np.array([0, 0, 1])
    V = np.array([0, 0, 1])
    L = np.array([np.sqrt(2)/2, 0, np.sqrt(2)/2])
    total, diff, spec = cook_torrance(N, V, L, 0.5, 0.5, 0.0)
    # 預期範圍 0.155 - 0.165
    assert 0.155 < total < 0.165, f"Example 16.2 failed: {total}"
    print(f"Test 4 (Example 16.2): PASSED (Total = {total:.4f}, Diff = {diff:.4f}, Spec = {spec:.4f})")
    
    print("--- 所有測試通過 ---")

if __name__ == "__main__":
    run_tests()
```
